@Service
@RequiredArgsConstructor
public class MusicServiceImpl extends ServiceImpl<MusicMapper, MusicEntity> implements IMusicService {

    private final FileUploadService fileUploadService;
    private final UserCollectionMapper userCollectionMapper;
    private final RedisUtil redisUtil;

    @Override
    @Transactional(rollbackFor = Exception.class)
    public MusicVO uploadMusic(MultipartFile file) throws Exception {
        Long userId = JwtUtil.LOGIN_USER_HANDLER.get();
        // 1. 上传音频文件到OSS
        String musicUrl = fileUploadService.uploadFile(file);

        // 2. 提取音频元数据
        File tempFile = Files.createTempFile("music-", file.getOriginalFilename()).toFile();
        file.transferTo(tempFile);

        AudioFile audioFile = AudioFileIO.read(tempFile);
        Tag tag = audioFile.getTag();
        int duration = audioFile.getAudioHeader().getTrackLength();

        String coverUrl = null;
        Artwork artwork = tag.getFirstArtwork();
        if (artwork != null) {
            byte[] imageData = artwork.getBinaryData();
            // 上传封面到OSS
            coverUrl = fileUploadService.uploadBytes(imageData, "cover-" + file.getOriginalFilename());
        }

        // 3. 构造MusicEntity并保存
        MusicEntity musicEntity = new MusicEntity()
                .setName(tag.getFirst(FieldKey.TITLE))
                .setArtistName(tag.getFirst(FieldKey.ARTIST))
                .setAlbumName(tag.getFirst(FieldKey.ALBUM))
                .setDuration(duration)
                .setMusicUrl(musicUrl)
                .setCoverUrl(coverUrl)
                .setUploaderId(userId)
                .setPlayCount(0L);
        this.save(musicEntity);

        // 清理临时文件
        tempFile.delete();

        return BeanUtil.copyProperties(musicEntity, MusicVO.class);
    }

    @Override
    public IPage<MusicVO> searchMusic(PageParam pageParam, String keyword) {
        Page<MusicEntity> page = new Page<>(pageParam.getCurrent(), pageParam.getSize());
        LambdaQueryWrapper<MusicEntity> wrapper = new LambdaQueryWrapper<>();
        if (StrUtil.isNotBlank(keyword)) {
            wrapper.like(MusicEntity::getName, keyword)
                    .or().like(MusicEntity::getArtistName, keyword)
                    .or().like(MusicEntity::getAlbumName, keyword);
        }
        wrapper.orderByDesc(MusicEntity::getCreateTime);
        IPage<MusicEntity> musicPage = this.page(page, wrapper);
        return PageUtils.convert(musicPage, MusicVO.class);
    }

    @Override
    public R collectMusic(Long musicId) {
        Long userId = JwtUtil.LOGIN_USER_HANDLER.get();
        // 检查是否已收藏
        long count = userCollectionMapper.selectCount(new LambdaQueryWrapper<UserCollectionEntity>()
                .eq(UserCollectionEntity::getUserId, userId)
                .eq(UserCollectionEntity::getMusicId, musicId));

        if (count > 0) {
            return R.failure("您已收藏该歌曲");
        }

        UserCollectionEntity collection = new UserCollectionEntity()
                .setUserId(userId)
                .setMusicId(musicId);
        userCollectionMapper.insert(collection);

        return R.success();
    }

    @Override
    public void uncollectMusic(Long musicId) {
        Long userId = JwtUtil.LOGIN_USER_HANDLER.get();
        userCollectionMapper.delete(new LambdaQueryWrapper<UserCollectionEntity>()
                .eq(UserCollectionEntity::getUserId, userId)
                .eq(UserCollectionEntity::getMusicId, musicId));
    }

    @Override
    public IPage<MusicVO> getMyCollections(PageParam pageParam) {
        Long userId = JwtUtil.LOGIN_USER_HANDLER.get();
        Page<UserCollectionEntity> page = new Page<>(pageParam.getCurrent(), pageParam.getSize());

        // 分页查询收藏关系
        IPage<UserCollectionEntity> collectionPage = userCollectionMapper.selectPage(page, new LambdaQueryWrapper<UserCollectionEntity>()
                .eq(UserCollectionEntity::getUserId, userId)
                .orderByDesc(UserCollectionEntity::getCreateTime));

        List<Long> musicIds = collectionPage.getRecords().stream()
                .map(UserCollectionEntity::getMusicId)
                .collect(Collectors.toList());

        if (CollUtil.isEmpty(musicIds)) {
            return new Page<>(pageParam.getCurrent(), pageParam.getSize(), 0);
        }

        // 查询歌曲详情
        List<MusicEntity> musicList = this.listByIds(musicIds);
        List<MusicVO> musicVOList = BeanUtil.copyToList(musicList, MusicVO.class);
        
        IPage<MusicVO> resultPage = new Page<>(collectionPage.getCurrent(), collectionPage.getSize(), collectionPage.getTotal());
        resultPage.setRecords(musicVOList);
        return resultPage;
    }

    private String getPlaybackQueueKey() {
        Long userId = JwtUtil.LOGIN_USER_HANDLER.get();
        return RedisKeys.PLAYBACK_QUEUE_KEY_PREFIX + userId;
    }
    
    @Override
    public List<MusicVO> getPlaybackList() {
        String key = getPlaybackQueueKey();
        List<Object> musicIdsObj = redisUtil.lGet(key, 0, -1);
        if (CollUtil.isEmpty(musicIdsObj)) {
            return Collections.emptyList();
        }
        List<Long> musicIds = musicIdsObj.stream()
            .map(id -> Long.valueOf(id.toString()))
            .collect(Collectors.toList());

        List<MusicEntity> musicList = this.listByIds(musicIds);
        return BeanUtil.copyToList(musicList, MusicVO.class);
    }

    @Override
    public void addToPlaybackList(Long musicId) {
        String key = getPlaybackQueueKey();
        redisUtil.lSet(key, musicId);
        redisUtil.expire(key, RedisKeys.PLAYBACK_QUEUE_TTL_DAYS, TimeUnit.DAYS);
    }

    @Override
    public void removeFromPlaybackList(Long musicId) {
        String key = getPlaybackQueueKey();
        // 移除所有等于musicId的项
        redisUtil.lRemove(key, 0, musicId);
    }
}
package com.hsx.manyue.modules.music.service.impl;

import cn.hutool.core.bean.BeanUtil;
import cn.hutool.core.collection.CollUtil;
import com.baomidou.mybatisplus.core.conditions.query.LambdaQueryWrapper;
import com.baomidou.mybatisplus.extension.service.impl.ServiceImpl;
import com.hsx.manyue.common.dto.R;

import lombok.RequiredArgsConstructor;
import org.springframework.stereotype.Service;
import org.springframework.transaction.annotation.Transactional;

import java.util.Collections;
import java.util.List;
import java.util.Objects;
import java.util.stream.Collectors;

@Service
@RequiredArgsConstructor
public class PlaylistServiceImpl extends ServiceImpl<PlaylistMapper, PlaylistEntity> implements IPlaylistService {

    private final PlaylistMusicMapper playlistMusicMapper;
    private final IMusicService musicService;

    @Override
    public PlaylistEntity createPlaylist(PlaylistCreateParam param) {
        Long userId = JwtUtil.LOGIN_USER_HANDLER.get();
        PlaylistEntity playlist = new PlaylistEntity()
                .setName(param.getName())
                .setDescription(param.getDescription())
                .setCreatorId(userId)
                .setSongCount(0);
        this.save(playlist);
        return playlist;
    }

    @Override
    @Transactional
    public R addMusicToPlaylist(Long playlistId, Long musicId) {
        Long userId = JwtUtil.LOGIN_USER_HANDLER.get();
        PlaylistEntity playlist = this.getById(playlistId);

        // 1. 检查歌单和权限
        if (playlist == null || !Objects.equals(playlist.getCreatorId(), userId)) {
            // 不抛异常，直接返回包含错误信息的 R 对象
            // 假设返回 403（无权限）或一个自定义业务错误码
            return R.failure("歌单不存在或您无权操作");
        }

        // 2. 检查歌曲是否已在歌单中
        long count = playlistMusicMapper.selectCount(new LambdaQueryWrapper<PlaylistMusicEntity>()
                .eq(PlaylistMusicEntity::getPlaylistId, playlistId)
                .eq(PlaylistMusicEntity::getMusicId, musicId));

        if (count > 0) {
            // 不抛异常，直接返回包含错误信息的 R 对象
            return R.failure("歌曲已在歌单中");
        }

        // 3. 执行成功操作
        playlistMusicMapper.insert(new PlaylistMusicEntity().setPlaylistId(playlistId).setMusicId(musicId));
        // 更新歌单歌曲数量
        this.update().setSql("song_count = song_count + 1").eq("id", playlistId).update();

        // 4. 返回成功的结果对象
        return R.success();
    }
    @Override
    @Transactional
    public void removeMusicFromPlaylist(Long playlistId, Long musicId) {
        Long userId = JwtUtil.LOGIN_USER_HANDLER.get();
        PlaylistEntity playlist = this.getById(playlistId);

        // 权限和歌单存在性检查
        if (playlist == null || !Objects.equals(playlist.getCreatorId(), userId)) {
            // 替换为 IllegalStateException
            throw new IllegalStateException("歌单不存在或您无权操作");
        }

        // 删除歌曲关联记录
        int deletedRows = playlistMusicMapper.delete(new LambdaQueryWrapper<PlaylistMusicEntity>()
                .eq(PlaylistMusicEntity::getPlaylistId, playlistId)
                .eq(PlaylistMusicEntity::getMusicId, musicId));

        // 如果成功删除了行数 (deletedRows > 0)，则更新歌单歌曲数量
        if (deletedRows > 0) {
            this.update().setSql("song_count = song_count - 1").eq("id", playlistId).update();
        }
    }

    @Override
    public PlaylistDetailVO getPlaylistDetail(Long playlistId) {
        PlaylistEntity playlist = this.getById(playlistId);
        if (playlist == null) {
            return null;
        }

        PlaylistDetailVO detailVO = BeanUtil.copyProperties(playlist, PlaylistDetailVO.class);

        // 查询歌单中的所有歌曲ID
        List<PlaylistMusicEntity> relations = playlistMusicMapper.selectList(new LambdaQueryWrapper<PlaylistMusicEntity>()
                .eq(PlaylistMusicEntity::getPlaylistId, playlistId)
                .orderByDesc(PlaylistMusicEntity::getCreateTime));
        
        if (CollUtil.isEmpty(relations)) {
            detailVO.setSongs(Collections.emptyList());
            return detailVO;
        }

        List<Long> musicIds = relations.stream().map(PlaylistMusicEntity::getMusicId).collect(Collectors.toList());
        List<MusicEntity> musicList = musicService.listByIds(musicIds);
        detailVO.setSongs(BeanUtil.copyToList(musicList, MusicVO.class));
        
        return detailVO;
    }
}

    }

    public static R failure(ReturnCodeEnums returnCode) {
        return result(returnCode);
    }

    public static R failure(String msg) {
        return new R().setCode(400000)
                .setMsg(msg);
    }

    private static R result(ReturnCodeEnums returnCode) {
        return new R().setCode(returnCode.getCode())
                .setMsg(returnCode.getMsg());
    }
